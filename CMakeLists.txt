cmake_minimum_required(VERSION 3.24)
project(ScanCleaner VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Quick QuickControls2 Pdf Concurrent)
include(CTest)
if(BUILD_TESTING)
    find_package(Qt6 6.4 REQUIRED COMPONENTS Test)
endif()
if(MSVC)
    add_compile_options(/utf-8)
endif()
find_package(OpenCV 4 REQUIRED COMPONENTS core imgproc)
qt_standard_project_setup()
if(Qt6_VERSION VERSION_GREATER_EQUAL "6.8")
    qt_policy(SET QTP0004 NEW)
endif()

qt_add_library(scan_processing STATIC src/Processing.cpp src/Processing.h)
target_include_directories(scan_processing PUBLIC src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(scan_processing PUBLIC Qt6::Gui Qt6::Pdf ${OpenCV_LIBS})
qt_add_executable(ScanCleaner src/main.cpp src/CleanerService.cpp src/CleanerService.h)
qt_add_qml_module(ScanCleaner URI ScanCleaner VERSION 1.0 RESOURCE_PREFIX /qt/qml
    QML_FILES qml/Main.qml qml/PageView.qml qml/ActionButton.qml)
target_link_libraries(ScanCleaner PRIVATE scan_processing Qt6::Quick Qt6::QuickControls2 Qt6::Concurrent)
target_compile_definitions(ScanCleaner PRIVATE SCAN_VERSION="${PROJECT_VERSION}")
option(SCAN_LEGACY_WINDOWS "Use the patched Qt SDK and software scene graph on Windows 7/8" OFF)
if(WIN32 AND SCAN_LEGACY_WINDOWS)
    target_compile_definitions(ScanCleaner PRIVATE SCAN_LEGACY_WINDOWS)
    if(MSVC)
        target_link_options(ScanCleaner PRIVATE /SUBSYSTEM:WINDOWS,6.01)
    endif()
endif()
set_target_properties(ScanCleaner PROPERTIES MACOSX_BUNDLE TRUE WIN32_EXECUTABLE TRUE
    MACOSX_BUNDLE_BUNDLE_NAME "净页"
    MACOSX_BUNDLE_GUI_IDENTIFIER "com.skyler.scan-cleaner"
    MACOSX_BUNDLE_SHORT_VERSION_STRING ${PROJECT_VERSION})
include(GNUInstallDirs)
install(TARGETS ScanCleaner BUNDLE DESTINATION . RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(APPLE AND Qt6_DIR MATCHES "^(/opt/homebrew|/usr/local/(opt|Cellar))/")
    string(CONFIGURE [=[
set(SCAN_QTPATHS "$<TARGET_FILE:Qt6::qtpaths>")
include("@CMAKE_CURRENT_SOURCE_DIR@/cmake/DeployMac.cmake")
]=] deploy_content @ONLY)
    qt_generate_deploy_script(TARGET ScanCleaner OUTPUT_SCRIPT deploy_script CONTENT "${deploy_content}")
    install(SCRIPT ${deploy_script})
elseif(APPLE)
    qt_generate_deploy_qml_app_script(TARGET ScanCleaner OUTPUT_SCRIPT deploy_script
        DEPLOY_TOOL_OPTIONS -codesign=- -no-strip)
    install(SCRIPT ${deploy_script})
elseif(WIN32)
    qt_generate_deploy_qml_app_script(TARGET ScanCleaner OUTPUT_SCRIPT deploy_script
        DEPLOY_TOOL_OPTIONS --no-compiler-runtime)
    install(SCRIPT ${deploy_script})
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    install(FILES packaging/linux/io.github.duhbbx.ScanCleaner.desktop
        DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
    install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATAROOTDIR}/licenses/scan-cleaner)
    set(SCAN_PACKAGE_FLAVOR "linux" CACHE STRING "Suffix for distribution-specific packages")
    set(CPACK_PACKAGE_NAME "scan-cleaner")
    set(CPACK_PACKAGE_CONTACT "ScanCleaner contributors <https://github.com/duhbbx/ScanCleaner>")
    set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Offline cleanup for scanned PDFs and images")
    set(CPACK_PACKAGE_FILE_NAME "ScanCleaner-${PROJECT_VERSION}-${SCAN_PACKAGE_FLAVOR}")
    set(CPACK_PACKAGING_INSTALL_PREFIX "/usr")
    set(CPACK_DEBIAN_PACKAGE_SECTION "graphics")
    set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
    set(CPACK_DEBIAN_PACKAGE_DEPENDS "qml6-module-qtquick, qml6-module-qtquick-controls, qml6-module-qtquick-dialogs, qml6-module-qtquick-layouts, qml6-module-qtquick-templates, qml6-module-qtquick-window, qml6-module-qtqml-workerscript, qt6-image-formats-plugins, qt6-qpa-plugins")
    include(CPack)
endif()
if(BUILD_TESTING)
    qt_add_executable(scan_tests tests/tst_processing.cpp)
    target_link_libraries(scan_tests PRIVATE scan_processing Qt6::Test)
    add_test(NAME scan_processing COMMAND scan_tests)
    set_tests_properties(scan_processing PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
